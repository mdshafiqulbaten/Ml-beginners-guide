 """Step 6: serve the model as a web API.         Run:   uvicorn app:app --reload"""
 import joblib
 import pandas as pd
 from fastapi import FastAPI
 from pydantic import BaseModel, Field

 # loaded ONCE at startup, not on every request
 bundle = joblib.load("churn_model.joblib")
 pipeline, threshold = bundle["pipeline"], bundle["threshold"]
 app = FastAPI(title="Churn predictor")

 class Customer(BaseModel):
     tenure_months: float = Field(ge=0, le=120)
     monthly_charges: float = Field(ge=0, le=500)
     contract: str
     internet_service: str
     payment_method: str
     support_calls: int = Field(ge=0, le=50)
     paperless_billing: int = Field(ge=0, le=1)
     total_charges: float | None = None

 @app.get("/health")
 def health():
     return {"status": "ok", "model_metrics": bundle["metrics"]}

 @app.post("/predict")
 def predict(customer: Customer):
     row = pd.DataFrame([customer.model_dump()])
     probability = float(pipeline.predict_proba(row)[0, 1])
     return {"churn_probability": round(probability, 3), "offer_retention_deal":
 probability >= threshold}
