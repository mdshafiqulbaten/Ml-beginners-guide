 """Copy-and-adapt template: a PyTorch training skeleton with validation, best-model
 saving and early stopping."""
 import copy
 import torch
 import torch.nn as nn
 from torch.utils.data import DataLoader, TensorDataset
 from sklearn.datasets import load_digits
 from sklearn.model_selection import train_test_split

 torch.manual_seed(0)
 device = "cuda" if torch.cuda.is_available() else "cpu"

 # DATA (replace with your own tensors)
 X, y = load_digits(return_X_y=True)
 X = torch.tensor(X / 16.0, dtype=torch.float32); y = torch.tensor(y)
 X_tr, X_va, y_tr, y_va = train_test_split(X, y, test_size=0.25, stratify=y,
 random_state=0)

train_dl = DataLoader(TensorDataset(X_tr, y_tr), batch_size=64, shuffle=True)
X_va, y_va = X_va.to(device), y_va.to(device)

# MODEL / LOSS / OPTIMIZER
model = nn.Sequential(nn.Linear(64, 128), nn.ReLU(), nn.Dropout(0.2), nn.Linear(128,
10)).to(device)
loss_fn, opt = nn.CrossEntropyLoss(), torch.optim.Adam(model.parameters(), lr=0.003)

best_loss, best_state, patience, bad = float("inf"), None, 5, 0
for epoch in range(100):
    model.train()
    for xb, yb in train_dl:
        xb, yb = xb.to(device), yb.to(device)
        opt.zero_grad(); loss_fn(model(xb), yb).backward(); opt.step()
    model.eval()
    with torch.no_grad():
        val_loss = loss_fn(model(X_va), y_va).item()
        acc = (model(X_va).argmax(1) == y_va).float().mean().item()
    if val_loss < best_loss:
        best_loss, best_state, bad = val_loss, copy.deepcopy(model.state_dict()), 0
    else:
        bad += 1
        if bad >= patience:
            print(f"early stop at epoch {epoch}"); break
model.load_state_dict(best_state)
torch.save(best_state, "best_model.pt")
print(f"best val loss {best_loss:.3f}, val accuracy {acc:.3f}")
